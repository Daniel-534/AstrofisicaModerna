\begin{figure}[H]
\centering
\begin{tikzpicture}[>=Latex,scale=0.98]
    \coordinate (S) at (-5.75,0.15);
    \coordinate (Sp) at (-1.55,0.78);
    \coordinate (Spp) at (3.65,0.78);

    % Estación y marco S
    \draw[gray!70!black,very thick] (-6.35,-0.35) -- (6.25,-0.35);
    \foreach \x in {-6.0,-5.4,...,5.8}
        \draw[gray!55,thick] (\x,-0.35) -- ++(-0.22,-0.3);

    \draw[->,black,very thick] (S) -- ++(1.65,0)
        node[right] {$x^1$};
    \draw[->,black,very thick] (S) -- ++(0,1.45)
        node[above] {$x^0$};
    \node[below=4pt] at (S) {$S$};
    \node[font=\small] at (-5.75,-0.95) {estación};

    % Tren y marco S'
    \draw[black,very thick,rounded corners=3pt,fill=gray!8]
        (-2.0,0.35) rectangle (5.55,1.8);
    \draw[black,very thick,fill=gray!30] (-0.9,0.12) circle (0.22);
    \draw[black,very thick,fill=gray!30] (4.55,0.12) circle (0.22);

    \draw[->,blue!75!black,very thick] (Sp) -- ++(1.55,0)
        node[right] {$x'^1$};
    \draw[->,blue!75!black,very thick] (Sp) -- ++(0,1.25)
        node[above] {$x'^0$};
    \node[blue!75!black,below=4pt] at (Sp) {$S'$};
    \node[blue!75!black,font=\small] at (-1.3,0.02) {tren};

    % Pelota y marco S''
    \draw[red!75!black,very thick,fill=red!10] (Spp) circle (0.18);
    \draw[->,red!75!black,thick] (Spp) -- ++(1.2,0)
        node[right] {$x''^1$};
    \draw[->,red!75!black,thick] (Spp) -- ++(0,1.0)
        node[above] {$x''^0$};
    \node[red!75!black,below=5pt] at (Spp) {$S''$};
    \node[red!75!black,font=\small] at (3.65,0.08) {pelota};

    % Velocidades relativas
    \draw[->,blue!75!black,very thick]
        (-5.45,3.05) -- (-1.3,3.05)
        node[midway,above] {$v$: velocidad de $S'$ respecto a $S$};

    \draw[->,red!75!black,very thick]
        (-1.15,2.38) -- (3.95,2.38)
        node[midway,above] {$v'$: velocidad de $S''$ respecto a $S'$};

    \draw[->,black,very thick]
        (-5.45,3.85) -- (3.95,3.85)
        node[midway,above] {$w=v''$: velocidad de $S''$ respecto a $S$};

    % Ley de composición
    \node[draw=blue!75!black,rounded corners=3pt,thick,
        fill=blue!4,inner sep=6pt] at (0,-1.4)
        {$\displaystyle
        w=\frac{v+v'}{1+\dfrac{vv'}{c^2}}$};
\end{tikzpicture}
\caption{Composición de tres marcos inerciales alineados. El tren, asociado con $S'$, se mueve con velocidad $v$ respecto a la estación $S$. La pelota, asociada con $S''$, tiene velocidad $v'$ respecto al tren y velocidad compuesta $w=v''$ respecto a la estación.}
\label{fig:tres-marcos}
\end{figure}
